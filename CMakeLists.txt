cmake_minimum_required(VERSION 3.21)
project(PostProjectNatron LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PostProject 0.7 REQUIRED CONFIG)
find_package(Python3 3.10 REQUIRED COMPONENTS Development.Module)
add_library(adapter STATIC src/association.cpp src/resolution.cpp)
set_target_properties(adapter PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(adapter PUBLIC src)
target_compile_features(adapter PUBLIC cxx_std_17)
target_link_libraries(adapter PUBLIC PostProject::postproject)
target_compile_options(adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_library(_postproject_natron MODULE src/module.cpp)
set_target_properties(_postproject_natron PROPERTIES PREFIX "")
target_compile_definitions(_postproject_natron PRIVATE Py_LIMITED_API=0x030A0000)
target_link_libraries(_postproject_natron PRIVATE adapter Python3::Module)
target_compile_options(_postproject_natron PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(native-contract tests/contract.cpp)
  target_link_libraries(native-contract PRIVATE adapter)
  target_compile_options(native-contract PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME native-contract COMMAND native-contract "${CMAKE_CURRENT_BINARY_DIR}/contract-fixture")
endif()
